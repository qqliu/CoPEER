117 | \begin{abstract}
118 | Let $G$  be a finite group and ${\rm
119 | cd}(G)$  will be the set of the degrees of the complex irreducible
120 | characters of $G$.  Also let ${\rm cod}(G)$ be  the set
121 | of codegrees of the irreducible characters of $G$. The Taketa problem conjectures if $G$ is solvable,  then ${\rm dl}(G) \leq |{\rm cd}(G)|$, where ${\rm dl}(G)$ is the derived length of $G$. In this note, we show that  ${\rm dl}(G) \leq |{\rm cod}(G)|$ in some cases and we conjecture that this inequality holds if $G$ is a finite solvable group.
122 | \end{abstract}

125 | \section{Introduction}
126 | Throughout this paper, $G$ will be a finite group and ${\rm
127 | cd}(G)$ will be the set of the degrees of the complex irreducible
128 | characters of $G$.  The number ${\rm cod(\chi)}=|G:{\rm
129 | Ker}(\chi)|/\chi(1)$ is called the codegree of $\chi$. Also the set
130 | of codegrees of the irreducible characters of $G$ is denoted by
131 | ${\rm cod}(G)$. This definition for codegrees first appeared in
132 | \cite{QWW}, where the authors use a graph-theoretic approach to
133 | compare the structure of a group with its set of codegrees. We follow \cite{Is} for group theoretic terminologies and notations
134 | not defined here.
135 | 
136 | Taketa showed  that the derived length  ${\rm dl}(G)$ of the group $G$ is bounded by the number $|{\rm cd}(G)|$, i.e., ${\rm dl}(G) \leq |{\rm cd}(G)|$ (see \cite[Corollary 5.13]{Is}). The Isaacs-Seitz conjecture asserts that the Taketa's inequality holds for solvable groups. This conjecture is also known as the Taketa problem or Taketa's inequality. There has been much work done regarding this conjecture in the last two decades. Moreover, T. Berger in \cite{B} shows that this conjecture is true for groups of odd order. For more results we refer the reader to \cite{A, IsK, YEG}.
137 | 
138 | By \cite[Lemma 3.1, Theorem 3.4 and Theorem 3.5]{ABGGH}, we know that if $|{\rm cod}(G)|\leq 3$, then ${\rm dl}(G) \leq |{\rm cod}(G)|$.  In this paper we show that if $G$ is a nilpotent group or has square free order then  ${\rm dl}(G) \leq |{\rm cod}(G)|$. Also we show that if ${\rm cod}(G)$ consists of power of $p$ then ${\rm dl}(G) < |{\rm cod}(G)|.$ Thus we may propose the  the following conjecture, inspired by Taketa's inequality:\\
139 | {\bf Conjecture:} Let $G$ be a finite solvable  group. Then  ${\rm dl}(G) \leq |{\rm cod}(G)|$.\\

158 | First we prove the conjecture for $p$-groups.
159 | \begin{lemma}\label{l1}
160 | Let $G$ be a $p$-group and $|G|=p^n$, where $p$ is prime and $n \geq  1$.  Then ${\rm dl}(G) \leq |{\rm cod}(G)|$.
161 | \end{lemma}
162 | \begin{proof}
163 | We prove the assertion by induction on $|G|$. If $n=1$ then $G \cong \mathbb{Z}_p$ and by \cite[Lemma 3.1]{ABGGH}  ${\rm dl}(G)=1 < |{\rm cod}(G)|$. Thus we may suppose that $n \geq 2$. Let $M$  and $N$ be two minimal normal subgroups of $G$.  Thus $M \cap N=1$ and so $G$ is isomorphic to a subgroup of $G/M \times G/N$. By induction hypotheses  we see that ${\rm dl}(G) \leq {\rm max}\{{\rm dl}(G/M), {\rm dl}(G/N)\} \leq {\rm max}\{|{\rm cod}(G/M)|, |{\rm cod}(G/N)|\} \leq |{\rm cod}(G)|$,  as wanted. Thus we may suppose that $G$ has a unique minimal normal subgroup, say $N$. Since $G$ is a $p$-group we have $N \cap Z(G) \neq 1$ and so $N\leq Z(G)$.  Now every subgroup of $N$ is a normal subgroup of $G$ and since $N$ is a minimal normal subgroup of $G$, it implies that  $N \cong \mathbb{Z}_p$.  If $N$ is the unique subgroup of order $p$ then by \cite[Theorem 5.46]{R}, either $G$ is cyclic or generalized quaternion. If $G$ is cyclic then ${\rm dl}(G)=1$ and so  ${\rm dl}(G) \leq |{\rm cod}(G)|$. Thus we may assume that $G$ is a generalized quaternion and so ${\rm dl}(G)=2\leq |{\rm cod}(G)|$. Therefore $G$ has another subgroup of order $p$, say $H$. Now $H \cap N =1$ and by induction hypotheses  ${\rm dl}(G) \leq {\rm max}\{{\rm dl}(G/N), {\rm dl}(G/H)\} \leq {\rm max}\{|{\rm cod}(G/N)|, |{\rm cod}(G/H)|\} \leq |{\rm cod}(G)|$.

174 |  \begin{theorem}\label{t1}
175 | Let $G$ be a nilpotent group. Then ${\rm dl}(G) \leq |{\rm cod}(G)|$.
176 | \end{theorem}
177 | \begin{proof}
178 | If $G$ is abelian then ${\rm dl}(G)=1 \leq |{\rm cod}(G)|$. Thus we may suppose that $G$ is not abelian.
179 | Since $G$ is nilpotent, it follows that $G$ is the direct product of its Sylow $p$-subgroups. Let $G=P_1 \times P_2 \times \cdots \times P_n$, where $P_i \in Syl_{p_i}(G)$ and $(|P_i|, |P_j|)=1$ for each $1 \leq i, j \leq n$. By \cite[Corollary 2.3]{X} we know that ${\rm cod}(G)={\rm cod}(P_1) \times \cdots \times {\rm cod}(P_n)$. Also  we know that ${\rm dl}(G) \leq {\rm max}\{ {\rm dl}(P_i) \mid 1\leq i \leq n \}$. Thus ${\rm dl}(G) \leq {\rm dl}(P_i)$ for some Sylow $P_i$, where $1 \leq i \leq n$. Now by Lemma \ref{l1}, ${\rm dl}(G) \leq {\rm dl}(P_i) \leq |{\rm cod}(P_i)| \leq |{\rm cod}(G)|.$
180 |   \end{proof}

182 |   \begin{theorem}\label{t2}
183 |  Let $G$ be a finite group and $p$ be a fixed prime. If ${\rm cod}(G)$ consists of power of $p$ then ${\rm dl}(G) < |{\rm cod}(G)|.$
184 | \end{theorem}
185 | \begin{proof}
186 |  By \cite[Theorem 1.1]{X} $G$ is a $p$-group or a Frobenious group whose order has exactly  two prime divisors. If $G$ is a $p$-group then by Lemma \ref{l1},  ${\rm dl}(G) \leq |{\rm cod}(G)|$.  Thus we may suppose that $G$ is a Frobenious group  and $|G|=p^mq^n$, where $m,n \geq 1$. Let $N$ be the kernel of $G$ of order $p^m$ and $H$ be the complement of $G$ of order $q^n$.  First suppose that  $N$ is a minimal normal subgroup of $G$. Thus $N$ is an abelian group and  by \cite[Lemma 3.1]{ABGGH}, we have ${\rm cod}(N)=\{1, p\}$.\\
187 |  {\bf Claim 1.} $|{\rm cod}(G/N)|+|{\rm cod}(N)| \leq |{\rm cod}(G)|.$\\
188 |  We know that ${\rm cod}(G/N) \subseteq {\rm cod} (G)$.  Let $\chi \in {\rm Irr}(G/N)$ and $\psi \in  {\rm Irr}(N)$. If ${\rm cod}(\chi)={\rm cod}(\psi^G)$ then $|G:{\rm ker}(\chi)|/\chi(1)=|G:{\rm ker}(\psi^G)|/\psi^G(1)$.  Now  $|G:{\rm ker}(\chi)|/\chi(1)=|G:{\rm ker}(\psi^G)|/\psi(1)|G:N|=|G:{\rm ker}(\psi^G)|/|G:N|=|N|/|{\rm ker}(\psi^G)|$.
189 |  Since $N$ is a $p$-group and ${\rm cod}(\chi)|q^n$ we have a contradiction. Now suppose that $\chi=1_N , \psi \in {\rm Irr}(N)$ such that ${\rm cod}(\psi)=p$ and ${\rm cod}(\chi) \neq {\rm cod}(\psi)$. If   ${\rm cod}(\chi^G) = {\rm cod}(\psi^G)$. Then $|G:{\rm ker}(\chi^G)|/\chi^G(1)=|G:{\rm ker}(\psi^G)|/\psi^G(1)$ and so $|G:{\rm ker}(\chi^G)|/|G:N|=|G:N|/|G:N|=1.$ Thus  ${\rm ker}(\chi^G)={\rm ker}(\psi^G)=N$ and so ${\rm cod}(\chi)={\rm cod}(\psi)$, a contradiction. Thus  the cliam is hold and  since by Lemma \ref{l1}, ${\rm dl}(G) \leq {\rm dl}(G/N)+{\rm dl}(N) \leq |{\rm cod}(G/N)|+|{\rm cod}(N)|$ it implies that    ${\rm dl}(G) < |{\rm cod}(G)|.$

193 | Now suppose that $N$ is not the minimal normal subgroup of $G$. Let $T$ be a minimal normal subgroup of $G$ and $T <  N$. In this case by induction on $|G|$ we prove that ${\rm dl}(G) < |{\rm cod}(G)|$. We know that the smallest Frobenious group has order $pq$, where $p>q$ and $q| p-1$.  Suppose that $F_{p, q}$ is the Frobenious group of order $pq$. By checking the character table of $F_{p,q}$ (see \cite[Theorem 25.10]{JL}), it is easy to see that ${\rm dl}(G) < |{\rm cod}(G)|$. Now suppose that $|G|>pq$. Since $G/T$ is a Frobenious group and $|G/T|<|G|$, by induction hypotheses we have ${\rm dl}(G/T) < |{\rm cod}(G/T)|.$ Now ${\rm dl}(G) \leq {\rm dl}(G/T)+{\rm dl}(T)< |{\rm cod}(G/T)|+1.$ and so ${\rm dl}(G) \leq |{\rm cod}(G/T)|.$\\
194 | {\bf Claim 2.} $|{\rm cod}(G/T)| <|{\rm cod}(G)|.$\\
195 | Suppose to contrary that ${\rm cod}(G/T) ={\rm cod}(G).$ Since ${\rm cod}(G/T) ={\rm cod}(G)$, it follows that ${\rm cod}(G|T) \subseteq {\rm cod}(G/T)$. If   ${\rm cod}(G|T)={\rm cod}(G/T)$ then ${\rm cod(G)}={\rm cod}(G|T)$, a contradiction. Thus we may suppose that  ${\rm cod}(G|T) \subset {\rm cod}(G/T)$.
196 |  We know that ${\rm cod}(G/N) \subseteq {\rm cod}(G/T).$ If ${\rm cod}(G/N) = {\rm cod}(G/T)$, then since $G/N$ is a $q$-group and $p| |G/T|$ we have a contradiction. Thus ${\rm cod}(G/N) \subset {\rm cod}(G/T).$  Also we have ${\rm cod}(G|N) \subseteq  {\rm cod}(G|T)$.  If   ${\rm cod}(G|N) \subset  {\rm cod}(G|T)$ then   ${\rm cod}(G/N) \cup {\rm cod}(G|N) \subset {\rm cod}(G)$, a contradiction.  Also  if ${\rm cod}(G|N)={\rm cod}(G|T)$ then ${\rm cod}(G|N) \subset  {\rm cod}(G/T)$. Also since ${\rm cod}(G/N) \subset  {\rm cod}(G/T)$ it follows that  ${\rm cod}(G) \subset  {\rm cod}(G/T)$, a contradiction. Thus the claim is hold and ${\rm dl}(G) < |{\rm cod}(G)|$, as wanted.

225 |   \begin{lemma}\label{l2}
226 | Suppose that $C$ is a class of finite solvable groups which is closed with respect to taking quotient. Suppose there exists a group in $C$, say $G$,  such that ${\rm dl}(G) \nleq {\rm cod}(G)$ and also $G$ has the smallest possible order. Then $G$ has a unique minimal normal subgroup, say $N$, such that ${\rm cod}(G)={\rm cod}(G/N)$.
227 | \end{lemma}
228 | \begin{proof}
229 | If $G$ is abelian group then $G \cong  \mathbb{Z}_p$ and so ${\rm dl}(G) \leq |{\rm cod}(G)|$, a contradiction. Thus  $G$ is nonabelian solvable group  and $G$ has minimal normal subgroup. Let $M$ and $N$ be two minimal normal subgroups of $G$. Thus $M \cap N=1$ and so $G$ is isomorphic to a subgroup of $G/M \times G/N$. By our assumption we see that ${\rm dl}(G) \leq {\rm max}\{{\rm dl}(G/M), {\rm dl}(G/N)\} \leq {\rm max}\{|{\rm cod}(G/M)|, |{\rm cod}(G/N)|\} \leq |{\rm cod}(G)|$, a contradiction. Thus $G$ has a unique minimal normal subgroup, say $N$. Since $G$ is
230 |    solvable, it follows that $N$ is abelian. Thus ${\rm dl}(G)\leq {\rm dl}(N)+{\rm dl}(G/N)=1+{\rm dl}(G/N) \leq 1+|{\rm cod}(G/N)| \leq1+|{\rm cod}(G)| \leq {\rm dl}(G)$. Thus $|{\rm cod}(G/N)|=|{\rm cod}(G)|$ and so ${\rm cod}(G)={\rm cod}(G/N)$, as wanted.
231 |  \end{proof}

233 |   \begin{theorem}\label{t4}
234 | Suppose that $G$ is a solvable group and for all $\chi , \psi \in {\rm Irr}(G)$, ${\rm ker}(\chi)={\rm ker}(\psi)$ if $\chi(1)=\psi(1)>1$. Then ${\rm dl}(G) \leq |{\rm cod}(G)|.$
235 | \end{theorem}
236 | \begin{proof}
237 | Suppose that $G$ is a minimal counter example to this theorem. By Lemma \ref{l2}, $G$ has a unique minimal normal subgroup $M$ and ${\rm cod}(G/M)={\rm cod}(G)$. If $G$ is abelian then ${\rm dl}(G) \leq |{\rm cod}(G)|$, as wanted. Thus $G^{'} \neq 1$ and so $M \subseteq G^{'}$. Therefore ${\rm cod}(G \mid M) \subseteq {\rm cod}(G \mid G^{'}) \subseteq {\rm cod}(G)-{1}$. Let $k \in {\rm cod}(G \mid M)$ such that $k \neq 1$. Thus there exists an irreducible character $\chi$ of $G$ such that $\chi(1)=k$ and $M \nsubseteq {\rm ker}(\chi)$. On the other hand, $k \in {\rm cod}(G)={\rm cod}(G/M)$ and so there exists an irreducible character $\psi$ of $G$ such that $\psi(1)=k$ and $M \subseteq {\rm ker}(\psi)$. By hypothesis ${\rm ker}(\chi)={\rm ker}(\psi)$, a contradiction.
238 |  \end{proof}

241 | Suppose that $G$ is a solvable group  in which  distinct nonlinear irreducible characters have distinct   ${\rm cod}(G)$. Then  ${\rm dl}(G) \leq |{\rm cod}(G)|.$
242 | \end{corollary}
